            // ---------- INGESTÃO ----------
            /**
             * PROCESSADOR INDUSTRIAL v2.3
             * Duas estratégias de parsing combinadas:
             *  1) tabular real: "ID  DATA  15 dezenas" (tab/espaço/vírgula), como exportado
             *     diretamente do site da Caixa em colagem de planilha.
             *  2) legado: blocos "Concurso N (dd/mm/aaaa)" seguidos das dezenas.
             * Os resultados de ambas são mesclados e deduplicados por ID.
             */
            processInput() {
                const raw = document.getElementById('raw-input').value;
                if (!raw || !raw.trim()) return;

                let found = [];
                try {
                    found = found.concat(this._parseTabular(raw));
                    found = found.concat(this._parseConcursoBlocks(raw));
                } catch (e) {
                    console.error(e);
                    return this.toast("FALHA NO MOTOR DE INGESTÃO!");
                }

                if (found.length === 0) {
                    return this.toast("FORMATO NÃO RECONHECIDO! Verifique se há 15 dezenas + data por concurso.");
                }

                // Mescla por ID (primeira ocorrência prevalece) e ordena por ID
                const byId = new Map();
                found.forEach(r => { if (!byId.has(r.id)) byId.set(r.id, r); });

                let processedCount = 0;
                [...byId.values()].forEach(r => {
                    if (this.db.history.some(x => x.id === r.id)) return;

                    // Conferência automática se houver pleito em aberto para este ID
                    if (this.db.pleito && String(this.db.pleito.targetID) === String(r.id)) {
                        this.runAudit(r.dzs, r.id);
                    }

                    this.db.history.push({ id: r.id, date: r.date, dzs: r.dzs });
                    processedCount++;
                });

                if (processedCount > 0) {
                    this.db.history.sort((a, b) => parseInt(b.id) - parseInt(a.id));
                    this.save(); this.refreshUI();
                    this.toast(`${processedCount} RESULTADO(S) INTEGRADO(S)!`);
                    document.getElementById('raw-input').value = '';
                } else {
                    this.toast("DADOS JÁ PRESENTES OU INVÁLIDOS.");
                }
            },

            _normalizeDzs(numArr) {
                const uniq = [...new Set(numArr.map(n => parseInt(n, 10)))]
                    .filter(n => n >= 1 && n <= 25)
                    .sort((a, b) => a - b)
                    .map(n => String(n).padStart(2, '0'));
                return uniq.length === 15 ? uniq : null;
            },

            _parseTabular(raw) {
                const out = [];
                const lines = raw.split(/\r?\n/);
                for (const line of lines) {
                    const dateMatch = line.match(/\d{2}\/\d{2}\/\d{4}/);
                    if (!dateMatch) continue;
                    const idMatch = line.match(/^\s*(\d{3,5})\b/);
                    if (!idMatch) continue;

                    let rest = line
                        .slice(idMatch[0].length)
                        .replace(dateMatch[0], '');
                    const nums = rest.match(/\d{1,2}/g);
                    if (!nums) continue;

                    const dzs = this._normalizeDzs(nums);
                    if (dzs) out.push({ id: idMatch[1], date: dateMatch[0], dzs });
                }
                return out;
            },

            _parseConcursoBlocks(raw) {
                const out = [];
                const regexHeader = /Concurso\s+(\d+)\s+\((\d{2}\/\d{2}\/\d{4})\)/gi;
                const matches = [];
                let match;
                while ((match = regexHeader.exec(raw)) !== null) {
                    matches.push({ id: match[1], date: match[2], start: match.index, headerLen: match[0].length });
                }
                for (let i = 0; i < matches.length; i++) {
                    const start = matches[i].start;
                    const end = (i + 1 < matches.length) ? matches[i + 1].start : raw.length;
                    let blockContent = raw.substring(start, end);
                    blockContent = blockContent.replace(/Concurso\s+\d+/i, '').replace(/\(\d{2}\/\d{2}\/\d{4}\)/, '');
                    const nums = blockContent.match(/\d{1,2}/g);
                    if (!nums) continue;
                    const dzs = this._normalizeDzs(nums);
                    if (dzs) out.push({ id: matches[i].id, date: matches[i].date, dzs });
                }
                return out;
            },

            // ---------- ANÁLISE ESTATÍSTICA (MOTOR REAL) ----------
            /**
             * Recalcula toda a matriz de recorrências a partir do histórico atual.
             * Retorna null se não houver histórico.
             */
            computeAnalysis() {
                const h = this.db.history; // indice 0 = mais recente
                if (h.length === 0) return null;

                const toSet = (dzs) => new Set(dzs.map(Number));
                const last = toSet(h[0].dzs);
                const magicAll = this.magicAll();

                const pares = [...last].filter(n => n % 2 === 0).length;
                const borda = [...last].filter(n => this.constants.borda.includes(n)).length;
                const miolo = [...last].filter(n => this.constants.miolo.includes(n)).length; // FIX v2.3
                const magia = [...last].filter(n => magicAll.includes(n)).length;

                let ancora = null, nucleo3 = null;
                if (h.length >= 2) {
                    const prev = toSet(h[1].dzs);
                    ancora = [...last].filter(n => prev.has(n)).sort((a, b) => a - b);
                }
                if (h.length >= 3) {
                    const prev = toSet(h[1].dzs);
                    const prev2 = toSet(h[2].dzs);
                    nucleo3 = [...last].filter(n => prev.has(n) && prev2.has(n)).sort((a, b) => a - b);
                }

                // Score ponderado: peso maior para concursos mais recentes (até 3 últimos)
                const weights = [3, 2, 1];
                const freq = {};
                for (let n = 1; n <= 25; n++) freq[n] = 0;
                for (let i = 0; i < Math.min(3, h.length); i++) {
                    const s = toSet(h[i].dzs);
                    const w = weights[i];
                    s.forEach(n => { freq[n] += w; });
                }
                const ancoraSet = new Set(ancora || []);
                const nucleoSet = new Set(nucleo3 || []);
                const score = {};
                for (let n = 1; n <= 25; n++) {
                    let b = 0;
                    if (magicAll.includes(n)) b += 1;
                    if (ancoraSet.has(n)) b += 2;
                    if (nucleoSet.has(n)) b += 2;
                    score[n] = freq[n] + b;
                }
                const ranking = Array.from({ length: 25 }, (_, i) => i + 1)
                    .sort((a, b) => score[b] - score[a] || a - b);

                return { h, pares, borda, miolo, magia, ancora, nucleo3, freq, score, ranking, magicAll };
            },

            // ---------- GERADOR COMBINATÓRIO (lazy, sem materializar todo o espaço) ----------
            *_combinations(arr, k) {
                const n = arr.length;
                if (k < 0 || k > n) return;
                const idx = Array.from({ length: k }, (_, i) => i);
                while (true) {
                    yield idx.map(i => arr[i]);
                    let i = k - 1;
                    while (i >= 0 && idx[i] === i + n - k) i--;
                    if (i < 0) return;
                    idx[i]++;
                    for (let j = i + 1; j < k; j++) idx[j] = idx[j - 1] + 1;
                }
            },

            _validaJogo(nums, paresMin, paresMax, magiaMin, magiaMax) {
                const magicAll = this.magicAll();
                const pares = nums.filter(n => n % 2 === 0).length;
                if (pares < paresMin || pares > paresMax) return false;
                const magia = nums.filter(n => magicAll.includes(n)).length;
                if (magia < magiaMin || magia > magiaMax) return false;
                return true;
            },

            /**
             * Gera nJogos jogos distintos de 15 dezenas = core + (faltam) escolhidas
             * do pool ordenado por score, respeitando janelas de pares/magia.
             * Relaxa progressivamente os filtros se não encontrar jogos suficientes,
             * garantindo que sempre retorna nJogos jogos (nunca trava, nunca fica vazio).
             */
            gerarJogos(core, poolOrdenado, faltam, nJogos) {
                const coreSet = new Set(core);
                const candidatos = poolOrdenado.filter(c => !coreSet.has(c));
                if (faltam < 0 || faltam > candidatos.length) {
                    // core maior que o necessário: corta o excesso mantendo os de maior score
                    faltam = Math.max(0, Math.min(faltam, candidatos.length));
                }

                const janelas = [
                    { pMin: 7, pMax: 8, mMin: 3, mMax: 5 },   // estrita (calibragem atual)
                    { pMin: 6, pMax: 9, mMin: 2, mMax: 6 },   // relaxada
                    { pMin: 0, pMax: 15, mMin: 0, mMax: 9 }   // sem filtro (garantia de saída)
                ];

                for (const j of janelas) {
                    const jogos = [];
                    const vistos = new Set();
                    let attempts = 0;
                    const MAX_ATTEMPTS = 400000;
                    for (const comb of this._combinations(candidatos, faltam)) {
                        attempts++;
                        if (attempts > MAX_ATTEMPTS) break;
                        const jogo = [...core, ...comb].sort((a, b) => a - b);
                        if (!this._validaJogo(jogo, j.pMin, j.pMax, j.mMin, j.mMax)) continue;
                        const key = jogo.join('-');
                        if (vistos.has(key)) continue;
                        vistos.add(key);
                        jogos.push(jogo);
                        if (jogos.length === nJogos) return jogos;
                    }
                    if (jogos.length > 0 && jogos.length < nJogos) {
                        // aproveita o que achou nesta janela e completa nas próximas
                        if (j === janelas[janelas.length - 1]) return jogos;
                    }
                }
                return [];
            },

                let cicloCore = (nucleo3 && nucleo3.length > 0) ? nucleo3.slice()
                    : (ancora ? ancora.slice(0, Math.min(6, ancora.length)) : ranking.slice(0, 6));

                // Núcleo "bom custo": ciclo + reforços de maior score até atingir 7 dezenas (se possível)
                const cicloSet = new Set(cicloCore);
                const reforcos = ranking.filter(n => !cicloSet.has(n));
                const bomCore = cicloCore.concat(reforcos.slice(0, Math.max(0, 7 - cicloCore.length)));

                const premium = this.gerarJogos(baseCore, ranking, 15 - baseCore.length, 1);
                const excelente = this.gerarJogos(baseCore, ranking, 15 - baseCore.length, 3);
                const bom = this.gerarJogos(bomCore, ranking, 15 - bomCore.length, 9);

                    </div>
                `;

                target.innerHTML = `
                    ${hRender('PREMIUM (1 JOGO)', 'var(--accent-purple)', this.tempGames.premium, `núcleo fixo: ${baseCore.length} dz`)}
                    ${hRender('EXCELENTE CUSTO (3 JOGOS)', 'var(--accent-blue)', this.tempGames.excelente, `núcleo fixo: ${baseCore.length} dz`)}
                    <div style="max-height:250px; overflow-y:auto; padding-right:15px;">
                        ${hRender('BOM CUSTO (9 JOGOS)', 'var(--accent-green)', this.tempGames.bom, `núcleo fixo: ${bomCore.length} dz`)}
                        ${hRender('RAZOÁVEL (27 JOGOS)', 'var(--accent-orange)', this.tempGames.razoavel, `núcleo fixo: ${cicloCore.length} dz`)}

            },

            /**
             * AUDIT SYSTEM v2.3
             * FIX: tabela de prêmios agora cobre 11 a 15 pontos (14 e 15 faltavam).
             */
            runAudit(result, id) {
                const p = this.db.pleito;
                if (!p) return;
                let log = { id, investment: 0, returns: 0, blocks: [] };

                Object.keys(p.groups).forEach(key => {
                    let sec = { name: key, games: [] };
                    p.groups[key].forEach(gameStr => {
                        const gameArr = gameStr.split(' ');
                        const hits = gameArr.filter(d => result.includes(d)).length;
                        const prize = this.constants.prizeTable[hits] || 0;
                        sec.games.push({ gameStr, hits, prize });
                        log.investment += 3.50;
                        log.returns += prize;
                    });
                    log.blocks.push(sec);
                });

                this.db.financial += (log.returns - log.investment);
                this.db.auditLog.unshift(log);
                this.db.pleito = null;
            },

                            <div class="fin-card" style="background:var(--border-ui)">
                                <span>Líquido CC</span>
                                <strong style="color:${(log.returns - log.investment) >= 0 ? 'var(--accent-green)' : 'var(--accent-red)'}">R$ ${(log.returns - log.investment).toFixed(2)}</strong>
                            </div>
                        </div>
                        <div class="audit-note">Valores de 14 e 15 pontos são estimativas de referência — o prêmio real varia por concurso (acumulado). Confira o valor oficial na Caixa.</div>
                    </div>
                `).join('');
            },

            // ---------- SISTEMA ----------
            exportDB() {
                const blob = new Blob([JSON.stringify(this.db)], { type: 'application/json' });
                const a = document.createElement('a'); a.href = URL.createObjectURL(blob);
                a.download = `LFPRO_V23_CORE_${Date.now()}.json`; a.click();
            },

            importDB() {
                const f = document.createElement('input'); f.type = 'file'; f.accept = '.json';
                f.onchange = e => {
                    const reader = new FileReader();
                    reader.onload = ev => {
                        try {
                            const parsed = JSON.parse(ev.target.result);
                            if (!parsed || !Array.isArray(parsed.history)) {
                                throw new Error('Schema inválido: campo "history" ausente ou não é array.');
                            }
                            this.db = {
                                history: parsed.history || [],
                                pleito: parsed.pleito || null,
                                financial: typeof parsed.financial === 'number' ? parsed.financial : 0,
                                auditLog: parsed.auditLog || [],
                                theme: parsed.theme || 'dark'
                            };
                            this.save(); this.refreshUI();
                            this.toast("BACKUP INTEGRADO!");
                        } catch (err) {
                            console.error(err);
                            this.toast("ARQUIVO DE BACKUP INVÁLIDO OU CORROMPIDO!");
                        }
                    };
                    reader.onerror = () => this.toast("FALHA AO LER O ARQUIVO!");
                    reader.readAsText(e.target.files[0]);
                };
                f.click();
            },

            reset() {
                if (confirm("🛑 ATENÇÃO: EXCLUIR TODO O HISTÓRICO E BALANÇO?")) {
                    localStorage.removeItem('LFPRO_V23_SINGULARITY');
                    this.db = { history: [], pleito: null, financial: 0, auditLog: [], theme: 'dark' };
                    this.save(); this.refreshUI();
                    this.toast("SISTEMA RESETADO.");
                }
            }
        };

        // INICIALIZAÇÃO CORE
        window.addEventListener('DOMContentLoaded', () => engine.init());
    </script>
